def atomic_write_json(path, data):
    tmp = path + ".tmp"
    with open(tmp, "w") as f:
        json.dump(data, f, indent=2)
        f.flush()
        os.fsync(f.fileno())
    os.replace(tmp, path)


def struct_dicts(atoms_list):
    return [AseAtomsAdaptor.get_structure(a).as_dict() for a in atoms_list]

        "pathway_key": PATHWAY_KEY,

        "checkpoint_timestamp": time.time(),
    }

class TrackingBFGS(BFGS):
    # Self-registers as TrackingBFGS.last_instance and self-attaches an
    # externally configured observer the moment MatCalc instantiates it
    # (matcalc calls self.optimizer(atoms) / self.optimizer(self.neb), i.e.
    # exactly one positional arg, which this subclass accepts naturally).
    # MatCalc remains the sole caller of .run(); nothing here calls it.
    last_instance = None
    active_observer = None

    def __init__(self, *args, **kwargs):
        super().__init__(*args, **kwargs)
        type(self).last_instance = self
        observer = type(self).active_observer
        if observer is not None:
            self.attach(lambda: observer(self), interval=1)


def run_matcalc_tracked(matcalc_call, observer):
    TrackingBFGS.last_instance = None
    TrackingBFGS.active_observer = observer
    try:
        result = matcalc_call()
    finally:
        TrackingBFGS.active_observer = None
    optimizer = TrackingBFGS.last_instance
    TrackingBFGS.last_instance = None
    if optimizer is None:
        raise RuntimeError(
            "MatCalc did not instantiate TrackingBFGS -- the installed MatCalc "
            "version did not accept/use the supplied optimizer class as expected; "
            "refusing to silently fall back to a raw ASE optimizer"
        )
    return result, optimizer


def determine_convergence(matcalc_result, optimizer, final_forces, threshold):
    matcalc_conv = None
    if isinstance(matcalc_result, dict):
        for key in ("converged", "is_converged"):
            if key in matcalc_result:
                matcalc_conv = bool(matcalc_result[key])
                break
    flat_forces = np.asarray(final_forces, dtype=float).reshape(-1)
    shaped_forces = flat_forces.reshape(-1, 3)
    fmax_value = float(np.linalg.norm(shaped_forces, axis=1).max())
    fmax_conv = bool(fmax_value < threshold)

    # optimizer.converged()'s expected forces-argument shape is not
    # consistent across installed ASE versions -- confirmed directly against
    # real installed copies of both: ASE 3.24.0's Optimizer.converged(forces)
    # delegates to OptimizableAtoms.converged(forces, fmax), which computes
    # a per-atom force norm along axis 1 and therefore requires the natural
    # (n_atoms, 3) shape; ASE 3.26.0's Optimizer.converged(gradient) instead
    # asserts gradient.ndim == 1 and requires the flattened shape. The two
    # are mutually incompatible, and different real FP environments on this
    # cluster are confirmed to have different ASE versions pinned. Try every
    # known calling convention, in an order that costs nothing extra once one
    # succeeds, and never assume a fixed shape is correct for whatever ASE
    # version any given FP environment actually has installed.
    optimizer_conv = None
    _shape_errors = (TypeError, ValueError, AssertionError)
    for _attempt in (
        lambda: optimizer.converged(shaped_forces),
        lambda: optimizer.converged(flat_forces),
        lambda: optimizer.converged(),
    ):
        try:
            optimizer_conv = bool(_attempt())
            break
        except _shape_errors:
            continue
    if optimizer_conv is None:
        # optimizer.converged() could not be called successfully in any
        # known form on this ASE version -- fall back to the independently
        # computed fmax-vs-threshold comparison rather than crash the whole
        # calculation over an ASE-version incompatibility we cannot resolve.
        optimizer_conv = fmax_conv

    if matcalc_conv is not None and matcalc_conv != optimizer_conv:
        raise RuntimeError(
            "MatCalc-reported convergence (%r) disagrees with the tracked "
            "optimizer convergence (%r); refusing to silently pick one"
            % (matcalc_conv, optimizer_conv)
        )
    if optimizer_conv != fmax_conv and abs(fmax_value - threshold) > 1e-12:
        raise RuntimeError(
            "tracked optimizer convergence (%r) disagrees with the fmax-threshold "
            "fallback (%r, fmax=%.6f, threshold=%.6f)"
            % (optimizer_conv, fmax_conv, fmax_value, threshold)
        )
    converged = matcalc_conv if matcalc_conv is not None else optimizer_conv
    return converged, fmax_value


def relax_endpoint(pmg_struct, tag):
    observed = {"n": 0}

    def _obs(opt):
        observed["n"] += 1
        # Defensive reshape: opt.atoms is a plain, unwrapped ase.Atoms object
        # here (RelaxCalc is called with relax_cell=False, so matcalc never
        # wraps it in a cell filter -- confirmed in both locally available
        # matcalc source generations), so get_forces() is contractually
        # (n_atoms, 3) per ASE's own Atoms API. A malformed 1D return has
        # been observed in practice from at least one FP calculator on a
        # real Zaratan job (AxisError: axis 1 is out of bounds for array of
        # dimension 1) -- reshape defensively rather than trust the
        # calculator's output shape, same pattern already used in _neb_obs.
        grad = np.asarray(opt.atoms.get_forces(), dtype=float).reshape(-1, 3)
        current_fmax = float(np.linalg.norm(grad, axis=1).max())
        write_runtime_checkpoint("endpoint_" + tag, opt.nsteps, observed["n"], current_fmax, None, None, [opt.atoms])

    relax_calc = RelaxCalc(calculator=calc, optimizer=TrackingBFGS, relax_cell=False,
                            fmax=RELAX_FMAX, max_steps=RELAX_MAX_STEPS)
    relax_result, optimizer = run_matcalc_tracked(lambda: relax_calc.calc(pmg_struct), _obs)
    endpoint_converged, final_fmax = determine_convergence(
        relax_result, optimizer, optimizer.atoms.get_forces(), RELAX_FMAX)
    relaxed_struct = AseAtomsAdaptor.get_structure(optimizer.atoms)
    return relaxed_struct, optimizer.nsteps, observed["n"], final_fmax, endpoint_converged


def find_working_autosort_tol(start_struct, end_struct, nimages):
    # Preflight only: this interpolation is never used as a substitute for
    # MatCalc's own interpolation/construction below -- only the winning
    # tolerance value is passed forward. Graceful fallback restored from the
    # old approved generator script's try_autosort_tols: if every tolerance
    # in AUTOSORT_TOLS fails to reliably match sites, return None (matching
    # the old script's behavior) rather than raising. NEBCalc.calc_images's
    # own autosort_tol default is falsy-checked by pymatgen (`if
    # autosort_tol:`), so None -- like 0 -- means "interpolate without
    # autosorting" (no site reordering), the same well-defined fallback the
    # old script relied on for these exact two pathways.
    for tol in AUTOSORT_TOLS:
        try:
            test_structs = start_struct.interpolate(
                end_struct, nimages=nimages + 1, interpolate_lattices=False,
                pbc=False, autosort_tol=tol,
            )
            if len(test_structs) != nimages + 2:
                raise RuntimeError("unexpected interpolation length %d" % len(test_structs))
            return tol
        except Exception:
            continue
    return None


def run_neb_tracked(start_struct, end_struct, nimages, tol, observer):
    if hasattr(NEBCalc, "calc_images"):
        neb_calc = NEBCalc(calculator=calc, optimizer=TrackingBFGS, climb=True,
                            fmax=NEB_FMAX, max_steps=NEB_MAX_STEPS)
        neb_result, optimizer = run_matcalc_tracked(
            lambda: neb_calc.calc_images(start_struct=start_struct, end_struct=end_struct,
                                          n_images=nimages, interpolate_lattices=False,
                                          autosort_tol=tol),
            observer,
        )
        neb = neb_calc.neb
    elif hasattr(NEBCalc, "from_end_images"):
        # Legacy MatCalc API: NEBCalc.__init__ constructs and instantiates
        # the optimizer immediately (self.optimizer = self.optimizer(self.neb)),
        # so TrackingBFGS must be captured around construction, not around
        # the later no-argument .calc(fmax=, max_steps=) call.
        TrackingBFGS.last_instance = None
        TrackingBFGS.active_observer = observer
        try:
            neb_calc = NEBCalc.from_end_images(
                start_struct, end_struct, calculator=calc, n_images=nimages,
                interpolate_lattices=False, autosort_tol=tol, optimizer=TrackingBFGS, climb=True,
            )
        finally:
            TrackingBFGS.active_observer = None
        optimizer = TrackingBFGS.last_instance
        TrackingBFGS.last_instance = None
        if optimizer is None:
            raise RuntimeError(
                "MatCalc did not instantiate TrackingBFGS during legacy NEBCalc "
                "construction; refusing to silently fall back to a raw ASE optimizer"
            )
        barrier, force = neb_calc.calc(fmax=NEB_FMAX, max_steps=NEB_MAX_STEPS)
        neb_result = {"barrier": barrier, "force": force}
        neb = neb_calc.neb
    else:
        raise RuntimeError(
            "installed matcalc.NEBCalc exposes neither calc_images nor "
            "from_end_images; cannot proceed without a silent raw-ASE fallback"
        )
    return neb_result, optimizer, neb


with open(INPUT_PATH) as f:

# coordinate can legitimately differ by more than 0.5 between start and end

# starting NEB force. No autosort_tol argument here: pymatgen's default
# autosort_tol=0 is falsy, so no site reordering happens -- this is a pure

# must run before relaxation: re-framing after relaxation is not equivalent
# if the relaxation itself crosses a cell face.
_interp = start_pmg.interpolate(end_pmg, nimages=NIMAGES + 1, pbc=True)

    print("  OK fmax=%.4f converged=%s endpoint_converged=%s steps=%d" % (
        final_fmax, neb_converged, endpoint_converged, optimizer_steps))
except Exception as e:
    result["final_fp_neb_images"] = {}
    result["neb_status"] = {
        "calculation_status": "failed",
        "neb_converged": None,
        "optimizer_steps": None,
        "log_entry_count": None,
        "last_fmax_eV_per_angstrom": None,
        "endpoint_converged": None,
        "endpoint_start_fmax_eV_per_angstrom": None,
        "endpoint_end_fmax_eV_per_angstrom": None,
        "endpoint_start_steps": None,
        "endpoint_end_steps": None,
        "autosort_tol_used": None,
        "error": type(e).__name__ + ": " + str(e),
    }
    result["calculation_metadata"] = base_calculation_metadata()
    print("  ERR " + str(e))

atomic_write_json(OUTPUT_PATH, result)
print("END " + OUTPUT_KEY + " " + PATHWAY_KEY + " -> " + OUTPUT_PATH)

def build_full_fp_neb_script(pot, reg_key, pathway_key, input_path, output_path, runtime_checkpoint_path):
    mp = repr(pot['model_path']) if pot['model_path'] else 'None'
    return (
        FULL_FP_NEB_TEMPLATE
        .replace('__SITE_PKGS__', pot['site_pkgs'])
        .replace('__IMPORT_LINES__', pot['import_lines'])
        .replace('__MODEL_PATH__', mp)
        .replace('__CALC_LINES__', pot['calc_lines'])
        .replace('__OUTPUT_KEY__', pot['output_key'])
        .replace('__PATHWAY_KEY__', pathway_key)
        .replace('__NIMAGES__', str(NIMAGES))
        .replace('__RELAX_FMAX__', str(RELAX_FMAX))
        .replace('__RELAX_MAX_STEPS__', str(RELAX_MAX_STEPS))
        .replace('__NEB_FMAX__', str(NEB_FMAX))
        .replace('__NEB_MAX_STEPS__', str(NEB_MAX_STEPS))
        .replace('__AUTOSORT_TOLS__', str(AUTOSORT_TOLS))
        .replace('__INPUT_PATH__', input_path)
        .replace('__OUTPUT_PATH__', output_path)
        .replace('__RUNTIME_CHECKPOINT_PATH__', runtime_checkpoint_path)
        .replace('__REG_KEY__', reg_key)
        .replace('__DISPLAY_NAME__', pot['display_name'])
        .replace('__ENV_PATH__', pot['venv_activate'])
        .replace('__SOURCE_REFERENCE_FILE__', str(DFT_REFERENCE_FILE))
        .replace('__SOURCE_REFERENCE_SHA256__', _dft_reference_sha256)
    )
